# Přesměrování HTTP na HTTPS
RewriteEngine On
RewriteBase /
RewriteCond %{HTTPS} !on
RewriteRule (.*) https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]


<IfModule mod_headers.c>

    # Pravidla pro CSP, nutné upravit při používání externích zdrojů - YT, Vimeo, GTM apod.
    Header always set Content-Security-Policy "\
        default-src 'self'; \
        script-src 'self'; \
        style-src 'self'; \
        img-src 'self' data:; \
        font-src 'self'; \
        connect-src 'self'; \
        media-src 'none'; \
        frame-src 'none'; \
        object-src 'none'; \
        base-uri 'none'; \
        form-action 'none'; \
        frame-ancestors 'none'; \
        upgrade-insecure-requests;"

    # Útočník si nemůže vložit můj web do iframe, starší fallback k CSP frame-ancestors
    Header always set X-Frame-Options "DENY"

    # Prohlížeč se nesnaží jinak interpretovat chybný content-type
    Header always set X-Content-Type-Options "nosniff"

    # Pravidla HSTS, neposílej zákazníka vůbec na HTTP, ale po první návštěvě rovnou na HTTPS
    Header always set Strict-Transport-Security "max-age=31536000"

    # Omezení informací o původní URL při přechodu na jiné weby
    Header always set Referrer-Policy "strict-origin-when-cross-origin"

</IfModule>
